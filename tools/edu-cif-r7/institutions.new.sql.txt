SELECT I.EDUCATION_INSTITUTIONS_ID AS ID_VALUE, I.INSTITUTION_CODE AS CODE_VALUE, CASE WHEN P.EDUCATION_INSTITUTIONS_ID IS NULL THEN I.NAME_FA ELSE P.NAME_FA || ' / ' || I.NAME_FA END AS LABEL_VALUE FROM CIF.EDUCATION_INSTITUTIONS I LEFT JOIN CIF.EDUCATION_INSTITUTIONS P ON P.EDUCATION_INSTITUTIONS_ID = I.PARENT_INSTITUTIONS_ID WHERE I.ACTIVE_FLAG = 'Y' AND I.INSTITUTION_STATUS_CODE IN ('ACTIVE','HISTORICAL','RENAMED','NEEDS_REVIEW') AND (:text IS NULL OR UPPER(I.INSTITUTION_CODE) LIKE :text OR UPPER(I.NAME_FA) LIKE :text OR UPPER(P.NAME_FA || ' / ' || I.NAME_FA) LIKE :text) ORDER BY LABEL_VALUE FETCH FIRST :rowLimit ROWS ONLY
