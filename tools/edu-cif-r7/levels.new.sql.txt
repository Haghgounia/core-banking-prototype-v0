SELECT EDUCATION_LEVELS_ID AS ID_VALUE, LEVEL_CODE AS CODE_VALUE, NAME_FA AS LABEL_VALUE FROM CIF.EDUCATION_LEVELS WHERE ACTIVE_FLAG = 'Y' AND IS_SELECTABLE = 'Y' AND (:text IS NULL OR UPPER(LEVEL_CODE) LIKE :text OR UPPER(NAME_FA) LIKE :text) ORDER BY LEVEL_ORDER, NAME_FA FETCH FIRST :rowLimit ROWS ONLY
