SELECT EDUCATION_INSTITUTIONS_ID AS ID_VALUE, INSTITUTION_CODE AS CODE_VALUE, DISPLAY_NAME_FA AS LABEL_VALUE FROM EDU.V_EDUCATION_INSTITUTION_LOOKUP WHERE (:text IS NULL OR UPPER(INSTITUTION_CODE) LIKE :text OR UPPER(DISPLAY_NAME_FA) LIKE :text) ORDER BY DISPLAY_NAME_FA FETCH FIRST :rowLimit ROWS ONLY
