SELECT EDUCATION_FIELD_LEVELS_ID AS ID_VALUE, FIELD_CODE || ':' || LEVEL_CODE AS CODE_VALUE, FIELD_NAME_FA || ' / ' || LEVEL_NAME_FA AS LABEL_VALUE FROM EDU.V_EDUCATION_FIELD_LEVEL_LOOKUP WHERE (:text IS NULL OR UPPER(FIELD_CODE) LIKE :text OR UPPER(LEVEL_CODE) LIKE :text OR UPPER(FIELD_NAME_FA) LIKE :text OR UPPER(LEVEL_NAME_FA) LIKE :text) ORDER BY FIELD_NAME_FA, LEVEL_NAME_FA FETCH FIRST :rowLimit ROWS ONLY
