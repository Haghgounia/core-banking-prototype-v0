SELECT FL.EDUCATION_FIELD_LEVELS_ID AS ID_VALUE, F.FIELD_CODE || ':' || L.LEVEL_CODE AS CODE_VALUE, F.NAME_FA || ' / ' || L.NAME_FA AS LABEL_VALUE FROM CIF.EDUCATION_FIELD_LEVELS FL JOIN CIF.EDUCATION_FIELDS F ON F.EDUCATION_FIELDS_ID = FL.EDUCATION_FIELDS_ID JOIN CIF.EDUCATION_LEVELS L ON L.EDUCATION_LEVELS_ID = FL.EDUCATION_LEVELS_ID WHERE FL.ACTIVE_FLAG = 'Y' AND F.ACTIVE_FLAG = 'Y' AND F.IS_SELECTABLE = 'Y' AND L.ACTIVE_FLAG = 'Y' AND (:text IS NULL OR UPPER(F.FIELD_CODE) LIKE :text OR UPPER(L.LEVEL_CODE) LIKE :text OR UPPER(F.NAME_FA) LIKE :text OR UPPER(L.NAME_FA) LIKE :text) ORDER BY F.NAME_FA, L.NAME_FA FETCH FIRST :rowLimit ROWS ONLY
