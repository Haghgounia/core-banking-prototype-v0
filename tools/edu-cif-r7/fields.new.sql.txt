SELECT EDUCATION_FIELDS_ID AS ID_VALUE, FIELD_CODE AS CODE_VALUE, NAME_FA AS LABEL_VALUE FROM CIF.EDUCATION_FIELDS WHERE ACTIVE_FLAG = 'Y' AND IS_SELECTABLE = 'Y' AND (:text IS NULL OR UPPER(FIELD_CODE) LIKE :text OR UPPER(NAME_FA) LIKE :text) ORDER BY DISPLAY_ORDER, NAME_FA FETCH FIRST :rowLimit ROWS ONLY
